cmake_minimum_required(VERSION 3.16)
project(pigeon LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Jetson Xavier (Carmel = ARMv8.2): -mcpu=native picks NEON/dotprod. Override with -DPIGEON_NATIVE=OFF for portable builds.
option(PIGEON_NATIVE "optimise for the build machine" ON)
if(PIGEON_NATIVE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    add_compile_options(-mcpu=native)
  else()
    add_compile_options(-march=native)
  endif()
endif()

# OpenCV: core imgproc video videoio dnn (+ bgsegm from opencv_contrib if present -> CNT background subtractor)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs video videoio dnn highgui OPTIONAL_COMPONENTS bgsegm)
message(STATUS "OpenCV ${OpenCV_VERSION}, bgsegm (CNT): ${OpenCV_bgsegm_FOUND}")

# TensorRT (JetPack ships it in /usr/include/aarch64-linux-gnu + /usr/lib/aarch64-linux-gnu)
option(PIGEON_WITH_TENSORRT "build the TensorRT backend if TensorRT is found" ON)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT install prefix (optional)")
if(PIGEON_WITH_TENSORRT)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu)
  find_library(TENSORRT_LIB nvinfer HINTS ${TENSORRT_ROOT}/lib /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu)
  find_package(CUDAToolkit QUIET)
  if(TENSORRT_INCLUDE_DIR AND TENSORRT_LIB AND CUDAToolkit_FOUND)
    message(STATUS "TensorRT: ${TENSORRT_LIB}")
  else()
    message(STATUS "TensorRT not found -> OpenCV DNN backend only")
    set(PIGEON_WITH_TENSORRT OFF)
  endif()
endif()

add_library(pigeon STATIC src/motion.cpp src/tracker.cpp src/models.cpp src/trt_backend.cpp src/runtime.cpp)
target_include_directories(pigeon PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pigeon PUBLIC ${OpenCV_LIBS} pthread)
if(PIGEON_WITH_TENSORRT)
  target_compile_definitions(pigeon PUBLIC PIGEON_WITH_TENSORRT)
  target_include_directories(pigeon PUBLIC ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(pigeon PUBLIC ${TENSORRT_LIB} CUDA::cudart)
endif()

add_executable(pigeon_detect apps/pigeon_detect.cpp)
target_link_libraries(pigeon_detect PRIVATE pigeon)

enable_testing()
add_executable(pigeon_tests tests/test_pigeon.cpp)
target_link_libraries(pigeon_tests PRIVATE pigeon)
add_test(NAME pigeon_tests COMMAND pigeon_tests WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/..)
